alter table pokemart_cupons enable row level security; -- sem policies: só as funções abaixo acessam create or replace function pokemart_is_admin(p_admin text) returns boolean language plpgsql security definer set search_path = public as $$ begin if lower(coalesce(p_admin,'')) in ('ricky','jef','lucas') then return true; end if; begin return exists (select 1 from admin_list() a where a.nome = p_admin); exception when others then return false; end; end $$; create or replace function pokemart_cupom_check(p_codigo text) returns table(codigo text, tipo text, valor int) language sql security definer set search_path = public as $$ select c.codigo, c.tipo, c.valor from pokemart_cupons c where c.codigo = upper(trim(p_codigo)) and c.ativo and (c.expira_em is null or c.expira_em >= current_date) and (c.usos_max is null or c.usos < c.usos_max) $$; create or replace function pokemart_cupom_usar(p_codigo text) returns void language sql security definer set search_path = public as $$ update pokemart_cupons set usos = usos + 1 where codigo = upper(trim(p_codigo)) and ativo and (usos_max is null or usos < usos_max) $$; create or replace function pokemart_cupom_listar(p_admin text) returns setof pokemart_cupons language plpgsql security definer set search_path = public as $$ begin if not pokemart_is_admin(p_admin) then raise exception 'Sem permissão.'; end if; return query select * from pokemart_cupons order by criado_em desc; end $$; create or replace function pokemart_cupom_salvar(p_admin text, p_codigo text, p_tipo text, p_valor int, p_expira date, p_usos_max int, p_ativo boolean) returns void language plpgsql security definer set search_path = public as $$ begin if not pokemart_is_admin(p_admin) then raise exception 'Sem permissão.'; end if; insert into pokemart_cupons(codigo,tipo,valor,expira_em,usos_max,ativo) values (upper(trim(p_codigo)),p_tipo,p_valor,p_expira,p_usos_max,p_ativo) on conflict (codigo) do update set tipo=excluded.tipo, valor=excluded.valor, expira_em=excluded.expira_em, usos_max=excluded.usos_max, ativo=excluded.ativo; end $$; create or replace function pokemart_cupom_excluir(p_admin text, p_codigo text) returns void language plpgsql security definer set search_path = public as $$ begin if not pokemart_is_admin(p_admin) then raise exception 'Sem permissão.';
